    A = np.array([[4.0, 1.0], [2.0, 3.0]])

    valores, vectores = np.linalg.eig(A)

- Solución numérica de ecuaciones no lineales (Newton-Raphson):
    def newton_raphson(f, df, x0, tol=1e-7, max_iter=50):
        x = float(x0)
        for _ in range(max_iter):
            fx = f(x)
            if abs(fx) < tol:
                return x
            x = x - fx / df(x)
        return x

- Integración numérica (Regla del Trapecio y Simpson 1/3):
    x = np.linspace(a, b, n + 1)
    y = f(x)
    h = x[1] - x[0]
    I_trap = (h / 2.0) * (y[0] + 2.0 * np.sum(y[1:-1]) + y[-1])
